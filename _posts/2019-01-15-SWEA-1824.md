SW Expert Academy에서 [1824. 혁진이의 프로그램 검증](https://www.swexpertacademy.com/main/code/problem/problemDetail.do?contestProbId=AV4yLUiKDUoDFAUx&categoryId=AV4yLUiKDUoDFAUx&categoryType=CODE) 문제를 해결했다.

'?'가 랜덤 4방향으로 움직여서 처리하는 것이 까다로웠지만 스텍에 네 방향을 저장해서 끝까지 못찾았을 경우 pop해서 다시 찾아 진행하였다. 시간초과가 나서 해결하기위해 `dV`에 방문여부를 저장해서 재방문할 경우 다시 스텍에 추가하지 않도록 하였다.

- `i`는 이동 횟수를 저장한다. 50번 이상 이동하였을 경우는 못찾는 경우라고 생각하여 종료하였다.
- `v`에는 현재 프로그램 내용을 저장하고 `dV`는 '?'가 나왔을 때 방문 여부를 확인하기 위하여 선언한 배열이다.
- `move`에는 위치 `x`와 `y` 값을 `move_d`에는 방향인 `d`와 이동횟수 `i` 값을 저장하였다.
- `i`가 50보다 클 경우나 '?'를 재방문 하였을 경우에는 `move`가 비었는지 확인하고 비지 않았을 경우 `move`의 값으로 이동한다. `move`가 비었을 경우는 종료한다.



```c++
#include<iostream>
#include<vector>
#include<string>
using namespace std;

//방향
int dx[] = { 0, 0, 1, -1 };
int dy[] = { 1, -1, 0, 0 };

int main() {
	int T;
	cin >> T;

	for (int test_case = 1; test_case <= T; test_case++) {
		int r, c;
		cin >> r >> c;
		vector<vector<char>> v;
		//'?'방문한걸 저장하기 위한 배열
		vector<vector<int>> dV;
		for (int i = 0; i < r; i++) {
			string s;
			cin >> s;
			vector<char> temp;
			vector<int> temp2;
			for (int j = 0; j < s.length(); j++) {
				temp.push_back(s.at(j));
				temp2.push_back(-1);
			}
			v.push_back(temp);
			dV.push_back(temp2);
		}

		int flag = 0; //종료플래그
		int x = 0;
		int y = 0;
		int d = 2; //오른쪽방향
		int num = 0;
		int i = 0;

		vector<pair<int, int>> move;
		vector<pair<int, int>> move_d;
		while (true) {
			if (i >= 50) {
				if (!move.empty()) {
					x = move.back().first;
					y = move.back().second;
					d = move_d.back().first;
					i = move_d.back().second;
					move.pop_back();
					move_d.pop_back();
				}
				else break;
			}
			else if (v[y][x] == '>') {
				d = 2;
			}
			else if (v[y][x] == 'v') {
				d = 0;
			}
			else if (v[y][x] == '^') {
				d = 1;
			}
			else if (v[y][x] == '-') {
				if (num == 0)
					num = 15;
				else num--;
			}
			else if (v[y][x] == '_') {
				if (num == 0)
					d = 2;
				else d = 3;
			}
			else if (v[y][x] == '<') {
				d = 3;
			}
			else if (v[y][x] == '|') {
				if (num == 0)
					d = 0;
				else d = 1;
			}
			else if (v[y][x] == '?') {
				//처음 방문일 경우
				if (dV[y][x] == -1) {
					i++;
					for (int j = 0; j < 3; j++) {
						move.push_back(make_pair(x, y));
						move_d.push_back(make_pair(j, i));
					}
					d = 3;
					dV[y][x] = d;
				}
				//이미 방문했을경우
				else {
					if (move.empty())
						break;
					x = move.back().first;
					y = move.back().second;
					d = move_d.back().first;
					i = move_d.back().second;
					move.pop_back();
					move_d.pop_back();
				}
			}
			else if (v[y][x] == '.') {

			}
			else if (v[y][x] == '@') {
				flag = 1;
				break;
			}
			else if (v[y][x] == '+') {
				if (num == 15)
					num = 0;
				else num++;
			}
			else {
				num = v[y][x] - '0';
			}
			y += dy[d];
			y = (y + r) % r;
			x += dx[d];
			x = (x + c) % c;
			i++;
		}
		if (flag == 0) {
			cout << "#" << test_case << " " << "NO" << "\n";
		}
		else {
			cout << "#" << test_case << " " << "YES" << "\n";
		}
	}
}
```
